# 1. Khi còn Internet: thay ten-theo-de-yeu-cau ở cả hai dòng.
npx create-expo-app@latest ten-theo-de-yeu-cau --template blank --no-agents-md
cd ten-theo-de-yeu-cau
New-Item -ItemType Directory -Force -Path src/components, src/screens, src/navigation

# 2. Khi còn Internet, chạy nguyên dòng dưới để cài một lượt.
# Gồm Bottom Tab, Stack, AsyncStorage, SQLite, camera, vị trí và thông báo.
npx expo install @react-navigation/native@7 @react-navigation/bottom-tabs@7 @react-navigation/native-stack@7 react-native-screens react-native-safe-area-context @react-native-async-storage/async-storage expo-sqlite expo-camera expo-location expo-notifications

# 3. Chạy app với Expo Go trên LDPlayer (terminal trong thư mục project).
npm start
# Giữ terminal mở. Trong LDPlayer, mở Expo Go và chọn project nếu thấy.
# Nếu chưa thấy, chọn Enter URL manually rồi nhập địa chỉ exp://... từ terminal.
# Trước buổi thi, thử ngắt Internet và sửa một dòng chữ để kiểm tra app vẫn cập nhật.
