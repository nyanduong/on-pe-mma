# PROJECT MỚI — bước 1 và 2 chạy khi còn Internet.
# 1. Thay ten-theo-de-yeu-cau ở cả hai dòng bên dưới.
npx create-expo-app@latest ten-theo-de-yeu-cau --template blank --no-agents-md
cd ten-theo-de-yeu-cau
New-Item -ItemType Directory -Force -Path src/components, src/screens, src/data

# 2. Cài một lượt các gói của bộ đề ôn (kể cả camera, vị trí, thông báo đã cài dự phòng).
npx expo install expo-sqlite @react-native-async-storage/async-storage @react-navigation/native@7 @react-navigation/bottom-tabs@7 @react-navigation/native-stack@7 react-native-screens react-native-safe-area-context @react-native-community/datetimepicker react-native-svg expo-camera expo-location expo-notifications

# 3. Đợi cài thành công, kiểm tra phiên bản rồi chạy app.
npx expo install --check
npm start
# Mở Expo Go tương thích trên LDPlayer, chọn project hoặc nhập địa chỉ exp://... từ terminal.
# Trước buổi thi: ngắt Internet, giữ terminal mở, sửa một dòng chữ rồi lưu để thử reload.
